{#- Overrides Material's social partial to allow an `image:` field on a social entry (rendered as ), so the craftbot.live link can use the favicon. Entries with `icon:` render the SVG icon as usual. -#} {% macro split_and_get(value, sep, index) %} {%- if value.split is defined -%} {%- set parts = value.split(sep) -%} {%- else -%} {%- set parts = value | split(sep) -%} {%- endif -%} {{- parts[index] -}} {% endmacro %}
